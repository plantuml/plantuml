package net.sourceforge.plantuml.abel;

import net.sourceforge.plantuml.klimt.color.Colors;
import net.sourceforge.plantuml.klimt.creole.Display;
import net.sourceforge.plantuml.stereo.Stereotype;

// One "note right/left of Class::member" tip, attached to a single member.
//
// Several tips on the same side of the same class are grouped by
// CommandFactoryTipOnEntity into one shared internal Entity (see
// Entity#putTip / Entity#getTips), so the color and the stereotype of
// each tip have to travel with its own Display rather than live as a
// single field on that shared Entity -- otherwise the last tip parsed
// overwrites the color/stereotype of every other tip on the same side
// of the same class (see issue #2814).
public class Tip {

	private final Display display;
	private final Colors colors;
	private final Stereotype stereotype;

	public Tip(Display display, Colors colors, Stereotype stereotype) {
		this.display = display;
		this.colors = colors;
		this.stereotype = stereotype;
	}

	public Display getDisplay() {
		return display;
	}

	public Colors getColors() {
		return colors;
	}

	public Stereotype getStereotype() {
		return stereotype;
	}

}
