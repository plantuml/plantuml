package org.scilab.forge.jlatexmath;

/**
 * The space around a {@link TeXIcon}, in pixels (what {@code java.awt.Insets}
 * was).
 */
public class TeXInsets {

    public int top;
    public int left;
    public int bottom;
    public int right;

    public TeXInsets(int top, int left, int bottom, int right) {
        this.top = top;
        this.left = left;
        this.bottom = bottom;
        this.right = right;
    }
}
