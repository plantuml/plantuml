package net.sourceforge.plantuml.teavm.c;

import java.io.IOException;
import java.io.InputStream;

/**
 * The standard input of the C build, for {@code -pipe}.
 * <p>
 * {@code System.in} cannot be used there: in the classlib of the TeaVM C backend,
 * {@code ConsoleInputStream.read} always throws, so the process aborted before
 * reading the first byte of the source.
 * <p>
 * Only create it when {@code TeaVM.isC()}: elsewhere the native method fails.
 */
public final class StdinStream extends InputStream {
	// ::remove file when JAVA8

	// The native method fills the start of its array: go through a block
	private final byte[] block = new byte[8192];

	public StdinStream() {
	}

	@Override
	public int read() throws IOException {
		final byte[] one = new byte[1];
		final int count = read(one, 0, 1);
		return count <= 0 ? -1 : one[0] & 0xFF;
	}

	@Override
	public int read(byte[] buffer, int offset, int length) throws IOException {
		if (length == 0)
			return 0;

		final int count = NativeStdin.read(block, Math.min(length, block.length));
		if (count == -1)
			return -1;
		if (count < 0)
			throw new IOException("Cannot read the standard input");

		System.arraycopy(block, 0, buffer, offset, count);
		return count;
	}

}
