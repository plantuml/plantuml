package net.sourceforge.plantuml.cli;

import net.sourceforge.plantuml.teavm.TeaVM;
// ::comment when JAVA8
import net.sourceforge.plantuml.teavm.c.NativeExit;
// ::done

public final class Exit {

	private Exit() {
	}

	public static final String DISABLE_PROPERTY = "plantuml.noSystemExit";

	public static void exit(int status) {
		// ::comment when JAVA8
		if (TeaVM.isC()) {
			// No System.exit in the classlib of the TeaVM C backend
			System.out.flush();
			System.err.flush();
			NativeExit.exit(status);
			return;
		}
		// ::done

		if (Boolean.getBoolean(DISABLE_PROPERTY))
			throw new ExitRequest(status);

		System.exit(status);
	}

	public static final class ExitRequest extends RuntimeException {
		public final int status;

		public ExitRequest(int status) {
			super("System.exit(" + status + ") intercepted");
			this.status = status;
		}
	}
}
